# 数据安全与监控

## 一、数据加密

### 1.1 文件加密

```bash
# 使用OpenSSL加密文件
openssl enc -aes-256-cbc -salt -in sensitive_data.txt -out sensitive_data.enc

# 解密文件
openssl enc -d -aes-256-cbc -in sensitive_data.enc -out sensitive_data.txt

# 生成密钥对
ssh-keygen -t ed25519 -f mykey -N ""

# 使用GPG加密
gpg --encrypt --recipient user@example.com sensitive_data.txt
gpg --decrypt sensitive_data.txt.gpg
```

### 1.2 数据库加密

```sql
-- MySQL数据加密
CREATE TABLE users (
    id INT PRIMARY KEY,
    username VARCHAR(50),
    email VARCHAR(100),
    password VARBINARY(256)
);

-- 插入加密数据
INSERT INTO users (id, username, email, password)
VALUES (1, 'admin', 'admin@example.com', AES_ENCRYPT('password123', 'encryption_key'));

-- 查询解密数据
SELECT id, username, email, AES_DECRYPT(password, 'encryption_key') AS password FROM users;
```

## 二、备份策略

### 2.1 数据库备份脚本

```bash
#!/bin/bash
# database_backup.sh - 数据库备份脚本

DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/backup"
DB_NAME="mydatabase"
DB_USER="backup_user"
DB_PASSWORD="secure_password"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份数据库
mysqldump -u $DB_USER -p$DB_PASSWORD $DB_NAME | gzip > $BACKUP_DIR/$DB_NAME_$DATE.sql.gz

# 保留最近7天的备份
find $BACKUP_DIR -type f -name "*.sql.gz" -mtime +7 -delete

# 同步到远程服务器
rsync -avz $BACKUP_DIR/ backup@remote-server:/backup/

echo "Backup completed: $BACKUP_DIR/$DB_NAME_$DATE.sql.gz"
```

### 2.2 文件备份脚本

```bash
#!/bin/bash
# file_backup.sh - 文件备份脚本

DATE=$(date +%Y%m%d)
SOURCE_DIR="/var/www/html"
BACKUP_DIR="/backup/files"
BACKUP_FILE="files_backup_$DATE.tar.gz"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 压缩备份
tar -czf $BACKUP_DIR/$BACKUP_FILE $SOURCE_DIR

# 保留最近30天的备份
find $BACKUP_DIR -type f -name "*.tar.gz" -mtime +30 -delete

echo "File backup completed: $BACKUP_DIR/$BACKUP_FILE"
```

## 三、日志监控

### 3.1 查看系统日志

```bash
# 查看SSH登录日志
tail -f /var/log/auth.log

# 查看系统日志
tail -f /var/log/syslog

# 查看Nginx访问日志
tail -f /var/log/nginx/access.log

# 查看Nginx错误日志
tail -f /var/log/nginx/error.log
```

### 3.2 日志分析工具

```bash
# 安装logwatch
apt install logwatch -y

# 生成日志报告
logwatch --detail High --mailto admin@example.com --range today

# 安装goaccess（实时日志分析）
apt install goaccess -y

# 实时分析Nginx日志
goaccess /var/log/nginx/access.log -o /var/www/html/report.html --log-format=COMBINED
```

### 3.3 日志轮转配置

```bash
# 查看logrotate配置
cat /etc/logrotate.conf

# 创建自定义日志轮转配置
cat > /etc/logrotate.d/myapp << EOF
/var/log/myapp/*.log {
    daily
    missingok
    rotate 7
    compress
    delaycompress
    notifempty
    create 640 www-data www-data
    postrotate
        systemctl reload myapp
    endscript
}
EOF
```

## 四、入侵检测

### 4.1 使用Tripwire

```bash
# 安装Tripwire（文件完整性检测）
apt install tripwire -y

# 初始化配置
tripwire --init

# 检查系统完整性
tripwire --check

# 更新策略
tripwire --update --twrfile /var/lib/tripwire/report/$(hostname)-$(date +%Y%m%d).twr
```

### 4.2 使用AIDE

```bash
# 安装aide（高级入侵检测环境）
apt install aide -y

# 初始化数据库
aideinit

# 检查系统完整性
aide --check

# 更新数据库
aide --update
cp /var/lib/aide/aide.db.new /var/lib/aide/aide.db
```

### 4.3 使用fail2ban

```bash
# 安装fail2ban
apt install fail2ban -y

# 配置SSH防护
cat > /etc/fail2ban/jail.local << EOF
[sshd]
enabled = true
port = ssh
filter = sshd
logpath = /var/log/auth.log
maxretry = 3
bantime = 3600

[nginx-botsearch]
enabled = true
port = http,https
filter = nginx-botsearch
logpath = /var/log/nginx/access.log
maxretry = 3
bantime = 86400
EOF

systemctl restart fail2ban
systemctl enable fail2ban
```

### 5.4 创建安全检查脚本

```bash
#!/bin/bash
# security_audit.sh - 安全检查脚本

echo "=== 安全检查报告 ==="
echo ""

echo "1. 检查开放端口:"
netstat -tlnp | grep -v "127.0.0.1"

echo ""
echo "2. 检查用户列表:"
cat /etc/passwd | grep -v nologin | awk -F: '{print $1, $7}'

echo ""
echo "3. 检查最近登录:"
last -n 10

echo ""
echo "4. 检查sudoers语法:"
visudo -c

echo ""
echo "5. 检查SSH配置:"
grep -E "^(Port|PermitRootLogin|PasswordAuthentication)" /etc/ssh/sshd_config

echo ""
echo "6. 检查fail2ban状态:"
systemctl is-active fail2ban
```
